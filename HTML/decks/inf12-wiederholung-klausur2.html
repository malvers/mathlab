Eine Musteraufgabe

-- Wie viele Schueler je Kurs, nur Kurse mit mindestens 3,
-- absteigend nach Anzahl:
SELECT k.bezeichnung, COUNT(*) AS anzahl
FROM Kurs k
JOIN Belegung b ON k.knr = b.knr
GROUP BY k.bezeichnung
HAVING COUNT(*) >= 3
ORDER BY anzahl DESC;
