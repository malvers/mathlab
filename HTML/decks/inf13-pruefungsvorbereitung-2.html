Was gibt dieses Programm aus?

zahlen = [4, 7, 2, 9, 5]
max_wert = zahlen[0]
summe = 0

for z in zahlen:
    summe = summe + z
    if z > max_wert:
        max_wert = z

print(summe, max_wert, summe / len(zahlen))
