Supabase Keys

Setup: in Supabase „Anonymous Sign-ins" aktivieren, dann im SQL-Editor:
CREATE TABLE messages (id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY, created_at timestamptz DEFAULT now(), sender text NOT NULL, content text NOT NULL, author uuid DEFAULT auth.uid(), pubkey text, sig text);
ALTER TABLE messages ENABLE ROW LEVEL SECURITY;
CREATE POLICY "msg read" ON messages FOR SELECT TO authenticated USING (true);
CREATE POLICY "msg insert" ON messages FOR INSERT TO authenticated WITH CHECK (author = auth.uid());
CREATE POLICY "msg update" ON messages FOR UPDATE TO authenticated USING (author = auth.uid()) WITH CHECK (author = auth.uid());
CREATE POLICY "msg delete" ON messages FOR DELETE TO authenticated USING (author = auth.uid());

CREATE TABLE identities (name_id text PRIMARY KEY, pubkey text NOT NULL, owner uuid DEFAULT auth.uid(), created_at timestamptz DEFAULT now());
ALTER TABLE identities ENABLE ROW LEVEL SECURITY;
CREATE POLICY "id read" ON identities FOR SELECT TO authenticated USING (true);
CREATE POLICY "id insert" ON identities FOR INSERT TO authenticated WITH CHECK (owner = auth.uid());
CREATE POLICY "id delete" ON identities FOR DELETE TO authenticated USING (owner = auth.uid());
